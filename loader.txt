local function seasonalTheme()
	local month = tonumber(os.date("%m")) or 0
	local day = tonumber(os.date("%d")) or 0
	if month == 12 or month == 1 or (month == 2 and day <= 15) then return "NewYear" end
	local names = {"Crimson", "Aurora", "Neon", "Ocean"}
	return names[(tonumber(os.date("%j")) or 1) % #names + 1]
end

local ACTIVE_THEME = seasonalTheme()
local THEME_ORDER = {"Crimson", "NewYear", "Aurora", "Neon", "Ocean"}
for key, value in pairs(THEMES[ACTIVE_THEME]) do C[key] = value end

local function applyTheme(name, roots)
	local theme = THEMES[name]
	if not theme then return end
	local previous = {}
	for key, value in pairs(C) do previous[key] = value end
	for key, value in pairs(theme) do C[key] = value end
	ACTIVE_THEME = name

	for _, root in ipairs(roots or {}) do
		if root and root.Parent then
			for _, obj in ipairs(root:GetDescendants()) do
				if obj:IsA("Frame") or obj:IsA("TextButton") or obj:IsA("TextBox") then
					if obj.BackgroundColor3 == previous.bg then obj.BackgroundColor3 = C.bg
					elseif obj.BackgroundColor3 == previous.card then obj.BackgroundColor3 = C.card
					elseif obj.BackgroundColor3 == previous.bar then obj.BackgroundColor3 = C.bar
					elseif obj.BackgroundColor3 == previous.accent then obj.BackgroundColor3 = C.accent end
				elseif obj:IsA("TextLabel") then
					if obj.TextColor3 == previous.accent then obj.TextColor3 = C.accent
					elseif obj.TextColor3 == previous.dim then obj.TextColor3 = C.dim
					elseif obj.TextColor3 == previous.sub then obj.TextColor3 = C.sub end
				elseif obj:IsA("UIStroke") then
					if obj.Color == previous.accent or obj.Color == previous.accent2 then obj.Color = C.accent end
				elseif obj:IsA("UIGradient") then
					obj.Color = ColorSequence.new({
						ColorSequenceKeypoint.new(0, C.accent),
						ColorSequenceKeypoint.new(1, C.bg),
					})
				end
			end
		end
	end
end

local function nextTheme()
	local index = table.find(THEME_ORDER, ACTIVE_THEME) or 1
	return THEME_ORDER[index % #THEME_ORDER + 1]
end

local function safeSetClipboard(text)
	local ok = pcall(setclipboard, text)
	if not ok then pcall(function() clipboard.set(text) end) end
end

local function readClipboard()
	local ok, v = pcall(function() return getclipboard() end)
	if ok and type(v) == "string" then return v end
	ok, v = pcall(function() return readclipboard() end)
	if ok and type(v) == "string" then return v end
	ok, v = pcall(function() return clipboard.get() end)
	if ok and type(v) == "string" then return v end
	return nil
end

local function httpGet(url)
	local attempts = {
		function() return game:HttpGet(url, true) end,
		function() return game:HttpGet(url) end,
		function() return HttpGet(url) end,
		function()
			local r = request({Url = url, Method = "GET"})
			if type(r) == "table" then return r.Body end
		end,
		function()
			local r = http_request({Url = url, Method = "GET"})
			if type(r) == "table" then return r.Body end
		end,
		function()
			local r = syn.request({Url = url, Method = "GET"})
			if type(r) == "table" then return r.Body end
		end,
	}
	for _, fn in ipairs(attempts) do
		local ok, res = pcall(fn)
		if ok and type(res) == "string" and res ~= "" then return res end
	end
	return nil
end

local function decodeJSON(str)
	local ok, data = pcall(function() return HttpService:JSONDecode(str) end)
	if ok and type(data) == "table" then return data end
	return nil
end

local function normalizeKey(raw)
	local t = tostring(raw or "")
	t = t:match("^%s*(.-)%s*$") or t
	t = t:gsub("^[%s'\"`%[%]]+", ""):gsub("[%s'\"`%[%]]+$", "")
	return t
end

local function isValidKey(k)
	return k ~= "" and string.lower(k) == string.lower(CORRECT_KEY)
end

local function keyFileApiOk()
	return type(isfile) == "function"
		and type(readfile) == "function"
		and type(writefile) == "function"
end

local function nowTime()
	local ok, t = pcall(function() return os.time() end)
	if ok and type(t) == "number" and t > 0 then return t end
	local ok2, t2 = pcall(function() return workspace:GetServerTimeNow() end)
	if ok2 and type(t2) == "number" and t2 > 0 then return math.floor(t2) end
	return 0
end

local KEY_PATHS = {KEY_FILE, KEY_FOLDER .. "/" .. KEY_FILE}

local function saveKeyToFile(key)
	if not keyFileApiOk() then return false end
	local payload = tostring(nowTime()) .. "\n" .. normalizeKey(key)
	for _, path in ipairs(KEY_PATHS) do
		if path ~= KEY_FILE then pcall(function() makefolder(KEY_FOLDER) end) end
		local ok = pcall(function() writefile(path, payload) end)
		if ok then return true end
	end
	return false
end

local function loadSavedKey()
	if not keyFileApiOk() then return nil end
	local ok, raw = pcall(function()
		for _, path in ipairs(KEY_PATHS) do
			if isfile(path) then
				local content = readfile(path)
				if type(content) == "string" and content ~= "" then return content end
			end
		end
		return nil
	end)
	if not ok or type(raw) ~= "string" or raw == "" then return nil end

	local ts, key = raw:match("^(%d+)%s*\n(.+)$")
	if not key then
		key, ts = raw, 0
	end
	key = normalizeKey(key)
	if not isValidKey(key) then return nil end

	local savedAt = tonumber(ts) or 0
	local now = nowTime()
	local age = now - savedAt
	if age < 0 then age = 0 end
	if savedAt > 0 and age > KEY_VALID_DAYS * 86400 then return nil end
	if savedAt == 0 then saveKeyToFile(key) end
	return key
end

local function fetchNameViaApi(placeId)
	local uniStr = httpGet("https://apis.roblox.com/universes/v1/places/" .. tostring(placeId) .. "/universe")
	local uni = decodeJSON(uniStr)
	local uid = type(uni) == "table" and uni.universeId
	if not uid then return nil end
	local gStr = httpGet("https://games.roblox.com/v1/games?universeIds=" .. tostring(uid))
	local g = decodeJSON(gStr)
	if type(g) == "table" and type(g.data) == "table" and type(g.data[1]) == "table" then
		local n = g.data[1].Name
		if type(n) == "string" and n ~= "" then return n end
	end
	return nil
end

local function productInfoName(info)
	if type(info) == "table" and type(info.Name) == "string" and info.Name ~= "" then
		return info.Name
	end
	return nil
end

local function fetchNameViaMarket(placeId)
	local ok, info = pcall(function()
		return MarketplaceService:GetProductInfo(placeId, Enum.AssetDependencyType.Place)
	end)
	local n = ok and productInfoName(info)
	if n then return n end
	ok, info = pcall(function() return MarketplaceService:GetProductInfo(placeId) end)
	return ok and productInfoName(info) or nil
end

local function fetchRealName(placeId)
	local n = fetchNameViaMarket(placeId)
	if n then return n end
	local okApi, name = pcall(fetchNameViaApi, placeId)
	if okApi and name then return name end
	return nil
end

local function getPlaceName(placeId)
	local n = fetchRealName(placeId)
	if n then return n end
	if type(game.Name) == "string" and game.Name ~= "" then
		return game.Name
	end
	return "Unknown game"
end

local function getExecutorName()
	local ok, name = pcall(function() return identifyexecutor() end)
	if not ok or type(name) ~= "string" or name == "" then
		ok, name = pcall(function() return getexecutorname() end)
	end
	if ok and type(name) == "string" and name ~= "" then return name end
	return "Unknown"
end

local function maskValue(value)
	local s = tostring(value)
	return string.rep("*", math.min(math.max(#s, 6), 12))
end

local cachedMaxPlayers = nil
local maxPlayersRow = nil

local function maxPlayersText()
	if cachedMaxPlayers then return tostring(cachedMaxPlayers) end
	local probes = {
		function() return game.MaxPlayers end,
		function() return game.Players.MaxPlayers end,
		function() return game:GetAttribute("MaxPlayers") end,
	}
	for _, probe in ipairs(probes) do
		local ok, v = pcall(probe)
		if ok and type(v) == "number" and v > 0 then
			cachedMaxPlayers = v
			return tostring(v)
		end
	end
	return "Loading..."
end

local function applyMaxPlayers()
	if not cachedMaxPlayers or not maxPlayersRow then return end
	maxPlayersRow.value = tostring(cachedMaxPlayers)
	local lbl = maxPlayersRow.labelRef
	if lbl and lbl.Parent then lbl.Text = maxPlayersRow.value end
end

local function New(class, props)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		inst[k] = v
	end
	return inst
end

local function corner(inst, radius)

local chipState = {}

local function applyChipColors(chip, instant)
	local s = chipState[chip]
	if not s or not chip.Parent then return end
	local bg, tr
	if s.hovered and s.hover then
		bg, tr = s.hover, s.hoverTrans
	else
		bg, tr = s.base, s.baseTrans
	end
	if instant then
		chip.BackgroundColor3 = bg
		chip.BackgroundTransparency = tr
	else
		TweenService:Create(chip, TweenInfo.new(0.14), {
			BackgroundColor3 = bg, BackgroundTransparency = tr,
		}):Play()
	end
end

local function iconChip(parent, x, y, size, iconId, opts)
	opts = opts or {}
	local z = opts.z or 16
	local isButton = opts.button == true
	local tone = opts.tone and TONE[opts.tone]
	local baseBg = opts.bg or (tone and tone.plate) or C.chip
	local baseTrans = opts.bgTrans or 0.05
	local props = {
		Position = UDim2.new(0, x, 0, y),
		Size = UDim2.new(0, size, 0, size),
		BackgroundColor3 = baseBg,
		BackgroundTransparency = baseTrans,
		BorderSizePixel = 0,
		ZIndex = z,
		Parent = parent,
	}
	if isButton then
		props.Text = ""
		props.AutoButtonColor = false
	end
	local chip = New(isButton and "TextButton" or "Frame", props)
	chipState[chip] = {base = baseBg, baseTrans = baseTrans}
	corner(chip, opts.radius or math.max(6, math.floor(size * 0.3)))
	if opts.stroke ~= false then
		uiStroke(chip, opts.strokeColor or Color3.fromRGB(255, 255, 255), 1, opts.strokeTrans or 0.8)
	end
	local iconSize = math.max(10, math.floor(size * (opts.iconScale or 0.58)))
	local img = New("ImageLabel", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0.5, 0, 0.5, 0),
		Size = UDim2.new(0, iconSize, 0, iconSize),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Image = iconId or "",
		ImageColor3 = opts.tint or (tone and tone.glyph) or C.icon,
		ScaleType = Enum.ScaleType.Fit,
		ZIndex = z + 1,
		Parent = chip,
	})
	return chip, img
end

local function iconButton(parent, x, y, size, iconId, onClick, opts)
	opts = opts or {}
	local chip, img = iconChip(parent, x, y, size, iconId, {
		button = true, z = opts.z, radius = opts.radius,
		bg = opts.bg, bgTrans = opts.bgTrans, iconScale = opts.iconScale,
		strokeColor = opts.strokeColor, strokeTrans = opts.strokeTrans,
		tint = opts.tint, tone = opts.tone,
	})
	local s = chipState[chip]
	if s then
		s.hover = opts.hoverBg or s.base:Lerp(Color3.new(1, 1, 1), 0.32)
		s.hoverTrans = opts.hoverTrans or math.max(0, s.baseTrans * 0.4)
	end
	chip.MouseEnter:Connect(function()
		if s then s.hovered = true end
		applyChipColors(chip)
	end)
	chip.MouseLeave:Connect(function()
		if s then s.hovered = false end
		applyChipColors(chip)
		TweenService:Create(chip, TweenInfo.new(0.14, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Size = UDim2.new(0, size, 0, size)
		}):Play()
	end)
	chip.MouseButton1Down:Connect(function()
		TweenService:Create(chip, TweenInfo.new(0.05), {
			Size = UDim2.new(0, size - 3, 0, size - 3),
		}):Play()
	end)
	chip.MouseButton1Up:Connect(function()
		TweenService:Create(chip, TweenInfo.new(0.14, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Size = UDim2.new(0, size, 0, size),
		}):Play()
		if touchMode and s then
			task.delay(0.14, function()
				if chip.Parent and s.hovered then
					s.hovered = false
					applyChipColors(chip)
				end
			end)
		end
	end)
	if onClick then
		chip.MouseButton1Click:Connect(function()
			onClick(chip, img)
		end)
	end
	return chip, img
end

local function flashCopied(img, chip)
	if not img then return end
	local prevIcon = img.Image
	local prevBg = chip and chipState[chip] == nil and chip.BackgroundColor3 or nil
	img.Image = ICON.check
	if chip then
		chip.BackgroundColor3 = Color3.fromRGB(150, 255, 195)
	end
	task.delay(1.2, function()
		if img.Parent then img.Image = prevIcon end
		if chip and chip.Parent then
			if chipState[chip] then
				applyChipColors(chip)
			elseif prevBg then
				chip.BackgroundColor3 = prevBg
			end
		end
	end)
end

local function fadeGroup(root, delayTime, duration)
	duration = duration or 0.4
	local saved = {}

	local function visit(d)
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			saved[#saved + 1] = {inst = d, kind = "text", a = d.TextTransparency, b = d.TextStrokeTransparency}
			d.TextTransparency = 1
			d.TextStrokeTransparency = 1
		elseif d:IsA("ImageLabel") then
			saved[#saved + 1] = {inst = d, kind = "image", a = d.ImageTransparency}
			d.ImageTransparency = 1
		elseif d:IsA("Frame") and d.BackgroundTransparency < 1 then
			saved[#saved + 1] = {inst = d, kind = "bg", a = d.BackgroundTransparency}
			d.BackgroundTransparency = 1
		elseif d:IsA("UIStroke") then
			saved[#saved + 1] = {inst = d, kind = "stroke", a = d.Transparency}
			d.Transparency = 1
		end
		for _, c in ipairs(d:GetChildren()) do
			visit(c)
		end
	end
	if not pcall(visit, root) then
		for _, it in ipairs(saved) do
			local inst = it.inst
			if inst.Parent then
				if it.kind == "text" then
					inst.TextTransparency = it.a
					inst.TextStrokeTransparency = it.b
				elseif it.kind == "image" then
					inst.ImageTransparency = it.a
				elseif it.kind == "bg" then
					inst.BackgroundTransparency = it.a
				else
					inst.Transparency = it.a
				end
			end
		end
		warn("[Syniox] fadeGroup rolled back after an error")
		return
	end

	task.delay(delayTime or 0, function()
		local tw = TweenInfo.new(duration, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
		for _, it in ipairs(saved) do
			local inst = it.inst
			if inst.Parent then
				if it.kind == "text" then
					TweenService:Create(inst, tw, {TextTransparency = it.a, TextStrokeTransparency = it.b}):Play()
				elseif it.kind == "image" then
					TweenService:Create(inst, tw, {ImageTransparency = it.a}):Play()
				elseif it.kind == "bg" then
					TweenService:Create(inst, tw, {BackgroundTransparency = it.a}):Play()
				else
					TweenService:Create(inst, tw, {Transparency = it.a}):Play()
				end
			end
		end
	end)
end

local fitted = {}

local function screenSize()
	local ok, sz = pcall(function()
		local s = ScreenGui.AbsoluteSize
		if s.X > 50 and s.Y > 50 then return s end
		return nil
	end)
	if ok and sz then return sz end
	local ok2, vs = pcall(function()
		local v = workspace.CurrentCamera.ViewportSize
		if v.X > 50 and v.Y > 50 then return v end
		return nil
	end)
	if ok2 and vs then return vs end
	return Vector2.new(1280, 720)
end

local function fitFor(w, h, sz)
	local s = math.min((sz.X - 18) / w, (sz.Y - 18) / h, 1)
	if s ~= s or s <= 0 then s = 1 end
	if s < 0.34 then s = 0.34 end
	return s
end

local function applyFit(instant)
	local sz = screenSize()
	for _, e in ipairs(fitted) do
		local target = fitFor(e.w, e.h, sz)
		if target ~= e.cur then
			e.cur = target
			if instant then
				e.comp.Scale = target
			else
				TweenService:Create(e.comp, TweenInfo.new(0.4, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
					Scale = target,
				}):Play()
			end
		end
	end
end

local function attachFit(frame, w, h)
	local sc = New("UIScale", {Scale = 1, Parent = frame})
	local entry = {comp = sc, w = w, h = h, cur = 1}
	fitted[#fitted + 1] = entry
	applyFit(true)
	return entry
end

local function addGameRow(y, icon, caption, value, withCopy, masked)
	local rf = New("Frame", {
		Name = "GameRow",
		Position = UDim2.new(0, 12, 0, y),
		Size = UDim2.new(1, -24, 0, 34),
		BackgroundColor3 = Color3.fromRGB(40, 5, 13),
		BackgroundTransparency = 0.4,
		BorderSizePixel = 0,
		ZIndex = 20,
		Parent = GameCard,
	})
	corner(rf, 10)
	uiStroke(rf, Color3.fromRGB(120, 16, 36), 1, 0.6)
	iconChip(rf, 6, 7, 20, icon, {z = 21, radius = 6, iconScale = 0.62, tone = "red"})
	label(rf, {
		Position = UDim2.new(0, 32, 0, 4),
		Size = UDim2.new(0, 130, 0, 10),
		Text = caption,
		Font = Enum.Font.GothamBold,
		TextSize = 8,
		TextColor3 = C.dim,
		ZIndex = 22,
	})
	local val = label(rf, {
		Position = UDim2.new(0, 32, 0, 14),
		Size = UDim2.new(1, withCopy and -92 or -38, 0, 15),
		Text = tostring(value),
		Font = Enum.Font.Code,
		TextSize = 11,
		TextTruncate = Enum.TextTruncate.AtEnd,
		ZIndex = 22,
	})
	local entry = {labelRef = val, value = tostring(value), hidden = masked == true}
	if masked then val.Text = maskValue(value) end

	if withCopy then
		local rowW = gcW - 24
		local cw = 50
		local chipX = rowW - cw - 6
		iconButton(rf, chipX, 5, 24, ICON.copy, function(chip, img)
			safeSetClipboard(entry.value)
			flashCopied(img, chip)
			showToast(LoaderFrame, "Copied  —  " .. entry.value, "success")
		end, {z = 23, radius = 7, iconScale = 0.6, tone = "green"})
		local eyeChip, eyeImg = iconButton(rf, chipX + 28, 5, 24, ICON.eye, function(chip, img)
			entry.hidden = not entry.hidden
			val.Text = entry.hidden and maskValue(entry.value) or entry.value
			img.Image = entry.hidden and ICON.eyeOff or ICON.eye
			showToast(LoaderFrame, entry.hidden and "Value hidden" or "Value revealed", "info")
		end, {z = 23, radius = 7, iconScale = 0.6, tone = "blue"})
		if masked then eyeImg.Image = ICON.eyeOff end
	end
	table.insert(gameRows, entry)
	table.insert(gameRowFrames, rf)
	return entry
end

local function executeScript(data)
	if runningScript then return end
	runningScript = data
	local prevSub = LdrSubText
	setLdrStatus("Running: " .. data.name .. " ...")
	local ok, err = pcall(function()
		local src = httpGet(data.url)
		if not src or src == "" then error("Script could not be downloaded") end
		local fn = loadstring(src)
		if not fn then error("loadstring is not supported") end
		fn()
	end)
	if ok then
		setLdrStatus("Ready - " .. data.name .. " executed")
	else
		setLdrStatus("Error - script could not be executed")
		warn("[Syniox] " .. tostring(err))
		task.delay(2.6, function()
			if runningScript == nil then setLdrStatus(prevSub) end
		end)
	end
	runningScript = nil
	return ok
end

-- High resolution artwork from the Roblox thumbnails API (GameId = universe id).
-- Disabled by default: swapping the working rbxthumb placeholder for an external

task.spawn(function()
	if not USE_REMOTE_THUMBNAILS then return end
	local gid = safeGameNumber(function() return game.GameId end)
	if not gid then return end
	local url = nil
	for attempt = 1, 3 do
		local ok, data = pcall(function()
			return decodeJSON(httpGet("https://thumbnails.roblox.com/v1/games/multiget/thumbnails?universeIds="
				.. tostring(gid) .. "&countPerUniverse=1&size=768x432&format=Png&filter=TimeWindow&timeWindow=AllTime"))
		end)
		if ok and type(data) == "table" and type(data.data) == "table" and type(data.data[1]) == "table"
			and type(data.data[1].thumbnails) == "table" and type(data.data[1].thumbnails[1]) == "table" then
			local u = data.data[1].thumbnails[1].imageUrl
			if type(u) == "string" and u:match("^https://") then
				url = u
				break
			end
		end
		task.wait(2)
	end
	if url and GameImg.Parent then
		GameImg.Image = url
	end
end)

local function applyThumbImage(entry, url)
	if type(url) ~= "string" or url == "" then return false end
	if not url:match("^https://") then return false end
	if not entry or not entry.thumb or not entry.thumb.Parent then return false end
	-- Never blank the box: the current image is swapped in place, so if this
	-- url fails to render we still show what we had (or the fallback icon).
	entry.thumb.ImageColor3 = Color3.fromRGB(255, 255, 255)
	entry.thumb.Image = url
	entry.thumb.ImageTransparency = 0
	entry.thumbReady = true
	if entry.thumbFallback then entry.thumbFallback.Visible = false end
	return true
end

task.spawn(function()
	if not USE_REMOTE_THUMBNAILS then
		-- Keep the local rbxthumb images that are already on every row.
		warn("[Syniox] remote thumbnails disabled (USE_REMOTE_THUMBNAILS = false)")
		return
	end
	local ids, byPlace = {}, {}
	for _, e in ipairs(scriptButtons) do
		if e.data.placeId then
			local key = tostring(e.data.placeId)
			if not byPlace[key] then
				byPlace[key] = e
				table.insert(ids, key)
			end
		end
	end
	if #ids == 0 then return end

	local function fetchItems(url)
		local ok, data = pcall(function()
			return decodeJSON(httpGet(url))
		end)
		if ok and type(data) == "table" and type(data.data) == "table" then
			return data.data
		end
		return nil
	end

	local function readyCount()
		local n = 0
		for _, e in pairs(byPlace) do
			if e.thumbReady then n = n + 1 end
		end
		return n
	end

	for round = 1, 3 do
		-- Method 1: batched place game icons
		if readyCount() < #ids then
			local items = fetchItems("https://thumbnails.roblox.com/v1/places/gameicons?placeIds="
				.. table.concat(ids, ",")
				.. "&returnPolicy=PlaceHolder&size=512x512&format=Png&isCircular=false")
			if items then
				for _, item in ipairs(items) do
					local e = byPlace[tostring(item.targetId or item.placeId)]
					if e then applyThumbImage(e, item.imageUrl) end
				end
			end
		end

		-- Method 2: resolve universe ids, then batched game icons (per game)
		if readyCount() < #ids then
			local uniIds, back = {}, {}
			for pid, e in pairs(byPlace) do
				if not e.thumbReady then
					local ok, uni = pcall(function()
						return decodeJSON(httpGet("https://apis.roblox.com/universes/v1/places/" .. pid .. "/universe"))
					end)
					if ok and type(uni) == "table" and uni.universeId then
						local uid = tostring(uni.universeId)
						table.insert(uniIds, uid)
						back[uid] = pid
					end
				end
			end
			if #uniIds > 0 then
				local items = fetchItems("https://thumbnails.roblox.com/v1/games/icons?universeIds="
					.. table.concat(uniIds, ",")
					.. "&returnPolicy=PlaceHolder&size=512x512&format=Png&isCircular=false")
				if items then
					for _, item in ipairs(items) do
						local pid = back[tostring(item.targetId)]
						local e = pid and byPlace[pid]
						if e then applyThumbImage(e, item.imageUrl) end
					end
				end
			end
		end

		if readyCount() >= #ids then break end
		task.wait(2.5)
	end

	warn("[Syniox] game thumbnails loaded: " .. readyCount() .. "/" .. #ids)
end)

-- Thumbnail reliability -------------------------------------------------------
-- 17 rows request their rbxthumb at the same time and some of those requests
-- fail silently, which leaves a row blank. ContentProvider reports the state of
-- every image, so only the broken rows are re-requested - rows that already
-- loaded are never touched (no flicker). Broken rows keep the visible gamepad
-- fallback icon instead of an empty box.
local function thumbStatus(uri)
	local ok, st = pcall(function()
		return ContentProvider:GetAssetFetchStatus(uri)
	end)
	if ok then return st end
	return nil
end

local function isBrokenStatus(st)
	return st == Enum.AssetFetchStatus.Failure or st == Enum.AssetFetchStatus.TimedOut
end

local function preloadInBackground(uri)
	task.spawn(function()
		pcall(function()
			ContentProvider:PreloadAsync({uri})
		end)
	end)
end

local function resolveGameIconUri(placeId)
	local ok, uni = pcall(function()
		return decodeJSON(httpGet("https://apis.roblox.com/universes/v1/places/" .. tostring(placeId) .. "/universe"))
	end)
	local uid = ok and type(uni) == "table" and uni.universeId
	if not uid then return nil end
	return rbxThumb("GameIcon", tostring(uid), 256, 256)
end

task.spawn(function()
	-- probe first: if this executor/client cannot report fetch status, bail out
	-- and keep the plain behaviour
	local okProbe = pcall(function()
		return ContentProvider:GetAssetFetchStatus("rbxassetid://1")
	end)
	if not okProbe then return end

	task.wait(2)

	for attempt = 1, 4 do
		local broken = {}
		for _, e in ipairs(scriptButtons) do
			if e.thumbUri and e.thumb and e.thumb.Parent then
				local st = thumbStatus(e.thumbUri)
				-- None = no record yet (preload it), or nothing ever happened
				-- (after a few attempts that counts as broken as well)
				if isBrokenStatus(st) or (st == Enum.AssetFetchStatus.None and attempt >= 3) then
					broken[#broken + 1] = e
				elseif st == Enum.AssetFetchStatus.None then
					preloadInBackground(e.thumbUri)
				end
			end
		end

		if #broken == 0 then
			if attempt > 1 then warn("[Syniox] all game thumbnails loaded") end
			return
		end

		for _, e in ipairs(broken) do
			local uri = e.thumbUri
			-- last attempts: the place thumbnail never arrives, try the game icon
			if attempt >= 3 then
				local alt = resolveGameIconUri(e.data.placeId)
				if alt then
					local okLoad = pcall(function()
						ContentProvider:PreloadAsync({alt})
					end)
					if okLoad and thumbStatus(alt) == Enum.AssetFetchStatus.Success then
						uri = alt
						e.thumbUri = alt
					end
				end
			end

			warn("[Syniox] reloading thumbnail " .. attempt .. "/4: " .. e.name)
			e.thumb.Image = ""
			if e.thumbFallback then e.thumbFallback.Visible = true end
			local target = uri
			task.defer(function()
				if e.thumb and e.thumb.Parent then e.thumb.Image = target end
			end)
		end

		task.wait(2.5)
	end

	local left = {}
	for _, e in ipairs(scriptButtons) do
		if e.thumbUri and e.thumb and e.thumb.Parent and isBrokenStatus(thumbStatus(e.thumbUri)) then
			left[#left + 1] = e.name
		end
	end
	if #left > 0 then
		warn("[Syniox] thumbnails still failing: " .. table.concat(left, ", "))
	end
end)

local function playOpenAnim()

			end

			end

		task.delay(0.12, function()
			for i, r in ipairs(restored) do
				if r.inst.Parent then
					TweenService:Create(r.inst, TweenInfo.new(0.5, Enum.EasingStyle.Quint,
						Enum.EasingDirection.Out, 0, false, (i % 14) * 0.024), {
						TextTransparency = r.tr,
						TextStrokeTransparency = r.sr,
					}):Play()
				end
			end
		end)
	end)
	if not okFade then
		for _, r in ipairs(restored) do
			if r.inst.Parent then
				r.inst.TextTransparency = r.tr
				r.inst.TextStrokeTransparency = r.sr
			end
		end
		warn("[Syniox] loader entrance fell back to plain state")
	end

	TweenService:Create(LoaderFrame, TweenInfo.new(0.7, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
		Position = UDim2.new(0.5, 0, 0.5, 0),
		BackgroundTransparency = 0.1,
	}):Play()
	pcall(function()
		TweenService:Create(loaderFit.comp, TweenInfo.new(0.75, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Scale = loaderFit.cur,
		}):Play()
	end)

	task.delay(0.18, function()
		if ListCard.Parent then
			TweenService:Create(ListCard, TweenInfo.new(0.6, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
				Position = listBase,
			}):Play()
		end
	end)
	task.delay(0.28, function()
		if GameCard.Parent then
			TweenService:Create(GameCard, TweenInfo.new(0.6, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
				Position = gameBase,
			}):Play()
		end
	end)
end

local function shakeInput()
	local base = inputFrame.Position
	local seq = {10, -10, 7, -7, 4, -4, 0}
	for i, dx in ipairs(seq) do
		task.delay((i - 1) * 0.045, function()
			if inputFrame.Parent then
				inputFrame.Position = base + UDim2.new(0, dx, 0, 0)
			end
		end)
	end
end

local verifying = false

local function enterLoader()
	pcall(function()
		LoaderFrame.Visible = true
		LoaderFrame.Position = UDim2.new(0.5, 0, -0.7, 0)
		LoaderFrame.BackgroundTransparency = 0.4
		loaderFit.comp.Scale = loaderFit.cur * 0.93
	end)
	playOpenAnim()
end

local function doVerify()
	if verifying then return end

	local text = normalizeKey(KeyBox.Text)

	if text == "" then
		pcall(setStatus, "idle")
		pcall(shakeInput)
		pcall(showToast, KeyFrame, "Key field is empty", "error")
		return
	end

	verifying = true
	pcall(setStatus, "load")

	task.delay(0.45, function()
		local okRun, err = pcall(function()
			if isValidKey(text) then
				pcall(setStatus, "ok")
				pcall(saveKeyToFile, text)
				enterLoader()
				KeyFrame.Visible = false
				task.delay(0.65, function()
					pcall(showToast, LoaderFrame, "Verification successful - welcome " .. tostring(Player.DisplayName), "success")
					task.delay(2.6, function()
						pcall(showToast, LoaderFrame, "Key saved - you can skip this screen for " .. KEY_VALID_DAYS .. " days", "info")
					end)
				end)
			else
				pcall(setStatus, "bad")
				pcall(shakeInput)
				pcall(showToast, KeyFrame, "Invalid key, please try again", "error")
			end
		end)
		verifying = false
		if not okRun then
			warn("[Syniox] verify error: " .. tostring(err))
			if isValidKey(text) then
				pcall(function()
					LoaderFrame.Visible = true
					LoaderFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
					LoaderFrame.BackgroundTransparency = 0.1
					for _, d in ipairs(LoaderFrame:GetDescendants()) do
						if (d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox"))
							and d.TextTransparency > 0 then
							d.TextTransparency = 0
						end
					end
					KeyFrame.Visible = false
				end)
			end
		end
	end)
end

verifyBtn.MouseButton1Click:Connect(doVerify)
KeyBox.FocusLost:Connect(function(enterPressed)
	if enterPressed then doVerify() end
end)

local function trySavedKeyEntry()
	local ok, saved = pcall(loadSavedKey)
	if not (ok and saved) then return false end

	KeyFrame.Visible = false
	enterLoader()

	task.delay(0.8, function()
		pcall(showToast, LoaderFrame,
			"Welcome back - key remembered for " .. KEY_VALID_DAYS .. " days", "success")
	end)
	return true
end

if not trySavedKeyEntry() then
	KeyFrame.BackgroundTransparency = 1
	KeyFrame.Position = UDim2.new(0.5, 0, 0.42, 0)

	local profBase = ProfileCard.Position

	end

	local textFadeIndex = 0

		local inInfoRow = d.Parent and d.Parent.Name == "InfoRow"

			d.TextTransparency = 1
			d.TextStrokeTransparency = 1

			TweenService:Create(d, TweenInfo.new(0.55, Enum.EasingStyle.Quint, Enum.EasingDirection.Out, 0, false, delayT), {
				TextTransparency = tr,
				TextStrokeTransparency = sr,
			}):Play()
		end
	end
	TweenService:Create(KeyFrame, TweenInfo.new(0.75, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
		Position = UDim2.new(0.5, 0, 0.5, 0),
		BackgroundTransparency = 0.1,
	}):Play()
	pcall(function()
		TweenService:Create(keyFit.comp, TweenInfo.new(0.8, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Scale = keyFit.cur,
		}):Play()
	end)
	task.delay(0.32, function()
		if ProfileCard.Parent then
			TweenService:Create(ProfileCard, TweenInfo.new(0.65, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
				Position = profBase,
			}):Play()
		end
	end)
	task.delay(0.6, function()
		pcall(function() KeyBox:CaptureFocus() end)
	end)
end
